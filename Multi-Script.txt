local C = {
	window = Color3.fromRGB(12, 12, 12),
	panel = Color3.fromRGB(15, 15, 15),

local registeredStrokes = {}
local registeredFrames = {}

local function updateThemeColors()
	if isNexus1 then return end
	
	if activeThemeMode == "light" then
		C.window = Color3.fromRGB(240, 240, 240)
		C.panel = Color3.fromRGB(225, 225, 225)
		C.item = Color3.fromRGB(210, 210, 210)
		C.hover = Color3.fromRGB(195, 195, 195)
		C.text = Color3.fromRGB(20, 20, 20)
		C.dim = Color3.fromRGB(100, 100, 100)
	else
		C.window = Color3.fromRGB(12, 12, 12)
		C.panel = Color3.fromRGB(15, 15, 15)
		C.item = Color3.fromRGB(18, 18, 18)
		C.hover = Color3.fromRGB(25, 25, 25)
		C.text = Color3.fromRGB(0, 3, 203)
		C.dim = Color3.fromRGB(0, 2, 120)
	end

	local strokeColor = Color3.fromRGB(0, 3, 203)
	if activeStrokeMode == "apple" then
		strokeColor = Color3.fromRGB(255, 0, 0)
	elseif activeStrokeMode == "sky" then
		strokeColor = Color3.fromRGB(0, 3, 203)
	end

	for _, s in ipairs(registeredStrokes) do
		if s and s.Parent then
			if activeStrokeMode == "sunrise" or activeStrokeMode == "moonlit" then
				local grad = s:FindFirstChildOfClass("UIGradient")
				if not grad then
					grad = Instance.new("UIGradient")
					grad.Parent = s
				end
				if activeStrokeMode == "sunrise" then
					grad.Color = ColorSequence.new(Color3.fromRGB(255, 100, 0), Color3.fromRGB(255, 0, 0))
				else
					grad.Color = ColorSequence.new(Color3.fromRGB(0, 102, 255), Color3.fromRGB(128, 0, 255))
				end
				s.Color = Color3.new(1, 1, 1)
			else
				local grad = s:FindFirstChildOfClass("UIGradient")
				if grad then grad:Destroy() end
				s.Color = strokeColor
			end
		end
	end

	for _, f in ipairs(registeredFrames) do
		if f and f.Parent then
			if f:IsA("Frame") or f:IsA("ScrollingFrame") then
				if f.Name == "Main" then
					f.BackgroundColor3 = C.window
				elseif f.BackgroundTransparency < 1 then
					f.BackgroundColor3 = C.panel
				end
			end
		end
	end
end

if isNexus1 then

local FONT = Enum.Font.Code
local TSIZE = 13
local LEFT = Enum.TextXAlignment.Left
local CENTER = Enum.TextXAlignment.Center
local AUTO_Y = Enum.AutomaticSize.Y
local BY_ORDER = Enum.SortOrder.LayoutOrder

local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do o[k] = v end
	o.Parent = parent
	return o
end

local function stroke(parent, color)
	local s = new("UIStroke", {
		Color = color or C.border, Thickness = 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, parent)
	table.insert(registeredStrokes, s)
	return s
end

local function label(parent, text, props)
	local p = {
		BackgroundTransparency = 1, BorderSizePixel = 0, Font = FONT, TextSize = TSIZE,
		TextColor3 = C.text, Text = text, TextXAlignment = LEFT,
	}
	for k, v in pairs(props or {}) do p[k] = v end
	return new("TextLabel", p, parent)
end

local function isPress(input)
	return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
end

local function isMove(input)
	return input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch
end

local function fire(cb, ...)
	if type(cb) ~= "function" then return end
	local args = table.pack(...)
	task.spawn(function()
		pcall(cb, table.unpack(args, 1, args.n))
	end)
end

local function destroyOldCopies()
	local roots = { CoreGui }
	if typeof(gethui) == "function" then
		local okH, h = pcall(gethui)
		if okH and h then table.insert(roots, h) end
	end
	local pg = player:FindFirstChild("PlayerGui")
	if pg then table.insert(roots, pg) end
	for _, r in ipairs(roots) do
		pcall(function()
			local old = r:FindFirstChild("NovaHubUI")
			if old then old:Destroy() end
		end)
	end
end

function NetUI:Notify(o)
	if not (notifHolder and notifHolder.Parent) then return end
	local f = new("Frame", {
		Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y,
		BackgroundColor3 = C.panel, BorderSizePixel = 0,
	}, notifHolder)
	table.insert(registeredFrames, f)
	stroke(f, C.border)
	
	new("Frame", { Size = UDim2.new(0, 4, 1, 0), Position = UDim2.new(1, -4, 0, 0), BackgroundColor3 = C.accent, BorderSizePixel = 0 }, f)

	local inner = new("Frame", {
		Size = UDim2.new(1, -8, 0, 0), AutomaticSize = AUTO_Y, BackgroundTransparency = 1,
		Position = UDim2.fromOffset(6, 0),
	}, f)
	new("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = BY_ORDER }, inner)
	new("UIPadding", { PaddingLeft = UDim.new(0, 4), PaddingRight = UDim.new(0, 8), PaddingTop = UDim.new(0, 8), PaddingBottom = UDim.new(0, 8) }, inner)
	
	if o.Title and o.Title ~= "" then
		label(inner, tostring(o.Title), { Size = UDim2.new(1, 0, 0, 16), TextColor3 = C.text, LayoutOrder = 1 })
	end
	label(inner, tostring(o.Content or ""), {
		Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, TextWrapped = true,
		TextColor3 = C.text, TextYAlignment = Enum.TextYAlignment.Top, LayoutOrder = 2,
	})
	
	task.delay(o.Duration or 5, function()
		if f.Parent then f:Destroy() end
	end)
end

function NetUI:CreateWindow(opts)
	opts = opts or {}
	destroyOldCopies()

	local W, H = 460, 520
	local gui = new("ScreenGui", {
		Name = "NovaHubUI", ResetOnSpawn = false, IgnoreGuiInset = true,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling, DisplayOrder = 999,
	})
	local parented = false
	if typeof(gethui) == "function" then
		parented = pcall(function() gui.Parent = gethui() end) and gui.Parent ~= nil
	end
	if not parented then parented = pcall(function() gui.Parent = CoreGui end) and gui.Parent ~= nil end
	if not parented then gui.Parent = player:WaitForChild("PlayerGui") end

	local main = new("Frame", {
		Name = "Main", Size = UDim2.fromOffset(W, H),
		Position = UDim2.new(0.5, -W / 2, 0.5, -H / 2),
		BackgroundColor3 = C.window, BorderSizePixel = 0,
	}, gui)
	table.insert(registeredFrames, main)
	stroke(main, C.border)

	local titleBar = new("Frame", { Size = UDim2.new(1, 0, 0, 28), BackgroundTransparency = 1, BorderSizePixel = 0 }, main)
	label(titleBar, opts.Name or "NETsourcesgg Hub", { Size = UDim2.new(1, -40, 1, 0), Position = UDim2.fromOffset(10, 0), TextColor3 = C.text })
	local closeBtn = new("TextButton", {
		Text = "x", Font = FONT, TextSize = 16, TextColor3 = C.text, AutoButtonColor = false,
		BackgroundTransparency = 1, Size = UDim2.fromOffset(28, 28), Position = UDim2.new(1, -28, 0, 0),
	}, titleBar)
	
	new("Frame", { Size = UDim2.new(1, 0, 0, 1), Position = UDim2.fromOffset(0, 28), BackgroundColor3 = C.border, BorderSizePixel = 0 }, main)

	local tabBar = new("Frame", { Size = UDim2.new(1, -16, 0, 24), Position = UDim2.fromOffset(8, 35), BackgroundTransparency = 1 }, main)
	new("UIListLayout", { FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 4), SortOrder = BY_ORDER }, tabBar)

	local pages = new("Frame", {
		Name = "PagesContainer", Size = UDim2.new(1, -16, 1, -72), Position = UDim2.fromOffset(8, 64),
		BackgroundTransparency = 1, ClipsDescendants = true,
	}, main)
	stroke(pages, C.border)

	local dragging, dragStart, startPos = false, nil, nil
	titleBar.InputBegan:Connect(function(input)
		if isPress(input) then dragging, dragStart, startPos = true, input.Position, main.Position end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and isMove(input) then
			local d = input.Position - dragStart
			main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
	UserInputService.InputEnded:Connect(function(input) if isPress(input) then dragging = false end end)

	closeBtn.MouseButton1Click:Connect(function() main.Visible = not main.Visible end)
	UserInputService.InputBegan:Connect(function(input)
		if input.KeyCode == Enum.KeyCode.N and not UserInputService:GetFocusedTextBox() then
			main.Visible = not main.Visible
		end
	end)

	notifHolder = new("Frame", {
		Size = UDim2.new(0, 300, 0, 0), AutomaticSize = AUTO_Y, AnchorPoint = Vector2.new(1, 1),
		Position = UDim2.new(1, -12, 1, -12), BackgroundTransparency = 1,
	}, gui)
	new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = BY_ORDER }, notifHolder)

	local Window = { _tabs = {} }

	function Window:CreateTab(name)
		local tab = {}
		local textW = TextService:GetTextSize(name, TSIZE, FONT, Vector2.new(1000, 100)).X
		local btn = new("TextButton", {
			Text = name, Font = FONT, TextSize = TSIZE, TextColor3 = C.dim, AutoButtonColor = false,
			BackgroundTransparency = 1, BorderSizePixel = 0, Size = UDim2.fromOffset(textW + 10, 22), LayoutOrder = #Window._tabs + 1,
		}, tabBar)
		stroke(btn, C.border)

		local page = new("ScrollingFrame", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0,
			ScrollBarThickness = 2, ScrollBarImageColor3 = C.accent, CanvasSize = UDim2.new(), Visible = false,
		}, pages)
		local contentCol = new("Frame", { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, BackgroundTransparency = 1 }, page)
		new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = BY_ORDER }, contentCol)
		new("UIPadding", { PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6), PaddingTop = UDim.new(0, 6) }, contentCol)

		contentCol:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
			page.CanvasSize = UDim2.fromOffset(0, contentCol.AbsoluteSize.Y + 12)
		end)

		tab._set = function(on)
			page.Visible = on
			btn.BackgroundColor3 = on and C.hover or C.window
			btn.TextColor3 = on and C.text or C.dim
		end
		btn.MouseButton1Click:Connect(function()
			for _, t in ipairs(Window._tabs) do t._set(t == tab) end
		end)
		table.insert(Window._tabs, tab)
		if #Window._tabs == 1 then tab._set(true) end

		local current
		local function newGroup(title)
			local box = new("Frame", {
				Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, BackgroundColor3 = C.panel, BorderSizePixel = 0, LayoutOrder = #contentCol:GetChildren(),
			}, contentCol)
			table.insert(registeredFrames, box)
			stroke(box, C.border)
			new("UIListLayout", { SortOrder = BY_ORDER }, box)
			local head = label(box, title, { Size = UDim2.new(1, 0, 0, 20), LayoutOrder = 1, TextColor3 = C.text })
			new("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8) }, head)
			local content = new("Frame", { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, BackgroundTransparency = 1, LayoutOrder = 2 }, box)
			new("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = BY_ORDER }, content)
			new("UIPadding", { PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8), PaddingBottom = UDim.new(0, 8) }, content)
			current = content
		end

		local function add()
			if not current then newGroup(name) end
			return current
		end

		function tab:CreateSection(title) newGroup(tostring(title)) end
		function tab:CreateLabel(text)
			local l = label(add(), tostring(text), { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, TextWrapped = true, TextColor3 = C.dim, TextYAlignment = Enum.TextYAlignment.Top })
			return { Set = function(_, t) l.Text = tostring(t) end }
		end

		function tab:CreateToggle(o)
			local value = o.CurrentValue and true or false
			local row = new("TextButton", { Text = "", AutoButtonColor = false, BackgroundTransparency = 1, Size = UDim2.new(1, 0, 0, 16) }, add())
			local box = new("Frame", { Size = UDim2.fromOffset(12, 12), Position = UDim2.fromOffset(0, 2), BackgroundColor3 = C.item, BorderSizePixel = 0 }, row)
			table.insert(registeredFrames, box)
			stroke(box, C.border)
			label(row, o.Name, { Position = UDim2.fromOffset(20, 0), Size = UDim2.new(1, -20, 1, 0) })
			local function render() box.BackgroundColor3 = value and C.accent or C.item end
			render()
			row.MouseButton1Click:Connect(function()
				value = not value
				render()
				fire(o.Callback, value)
			end)
			return { Set = function(_, v) value = v and true or false render() fire(o.Callback, value) end }
		end

		function tab:CreateSlider(o)
			local lo, hi, inc = o.Range[1], o.Range[2], o.Increment or 1
			local value = math.clamp(o.CurrentValue or lo, lo, hi)
			local holder = new("Frame", { Size = UDim2.new(1, 0, 0, 34), BackgroundTransparency = 1 }, add())
			label(holder, o.Name, { Size = UDim2.new(1, 0, 0, 14) })
			local bar = new("TextButton", { Text = "", AutoButtonColor = false, Position = UDim2.fromOffset(0, 16), Size = UDim2.new(1, 0, 0, 14), BackgroundColor3 = C.item, BorderSizePixel = 0 }, holder)
			table.insert(registeredFrames, bar)
			stroke(bar, C.border)
			local fill = new("Frame", { Size = UDim2.fromScale(0, 1), BackgroundColor3 = C.accent, BorderSizePixel = 0 }, bar)
			local txt = label(bar, "", { Size = UDim2.fromScale(1, 1), TextXAlignment = CENTER, ZIndex = 2, TextColor3 = Color3.new(1, 1, 1) })

			local function render()
				fill.Size = UDim2.fromScale((value - lo) / (hi - lo), 1)
				txt.Text = tostring(value) .. (o.Suffix or "")
			end
			render()

			local function setFromX(x)
				local rel = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
				local v = lo + math.floor(((lo + rel * (hi - lo)) - lo) / inc + 0.5) * inc
				value = math.clamp(v, lo, hi)
				render()
				fire(o.Callback, value)
			end
			local sliding = false
			bar.InputBegan:Connect(function(input) if isPress(input) then sliding = true setFromX(input.Position.X) end end)
			UserInputService.InputChanged:Connect(function(input) if sliding and isMove(input) then setFromX(input.Position.X) end end)
			UserInputService.InputEnded:Connect(function(input) if isPress(input) then sliding = false end end)
			return { Set = function(_, v) value = math.clamp(v, lo, hi) render() fire(o.Callback, value) end }
		end

		function tab:CreateButton(o)
			local b = new("TextButton", {
				Text = o.Name, Font = FONT, TextSize = TSIZE, TextColor3 = C.text, AutoButtonColor = false,
				BackgroundColor3 = C.item, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 20),
			}, add())
			table.insert(registeredFrames, b)
			local s = stroke(b, C.border)
			b.MouseEnter:Connect(function() b.BackgroundColor3 = C.hover end)
			b.MouseLeave:Connect(function() b.BackgroundColor3 = C.item s.Color = C.border end)
			b.MouseButton1Click:Connect(function() fire(o.Callback) end)
		end

		function tab:CreateDropdown(o)
			local options, selected = o.Options or {}, nil
			local holder = new("Frame", { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = AUTO_Y, BackgroundTransparency = 1 }, add())
			new("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = BY_ORDER }, holder)
			label(holder, o.Name, { Size = UDim2.new(1, 0, 0, 14), LayoutOrder = 1 })
			local head = new("TextButton", { Text = "", AutoButtonColor = false, BackgroundColor3 = C.item, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 18), LayoutOrder = 2 }, holder)
			table.insert(registeredFrames, head)
			stroke(head, C.border)
			local shown = label(head, "--", { Size = UDim2.new(1, -22, 1, 0), Position = UDim2.fromOffset(6, 0), TextTruncate = Enum.TextTruncate.AtEnd })
			local arrow = label(head, "v", { Size = UDim2.fromOffset(16, 18), Position = UDim2.new(1, -18, 0, 0), TextXAlignment = CENTER, TextColor3 = C.dim })
			local list = new("ScrollingFrame", { LayoutOrder = 3, Size = UDim2.new(1, 0, 0, 18), BackgroundColor3 = C.item, BorderSizePixel = 0, ScrollBarThickness = 2, AutomaticCanvasSize = AUTO_Y, Visible = false }, holder)
			table.insert(registeredFrames, list)
			stroke(list, C.border)
			new("UIListLayout", { SortOrder = BY_ORDER }, list)

			local function rebuild()
				if type(o.DynamicOptions) == "function" then options = o.DynamicOptions() end
				for _, ch in ipairs(list:GetChildren()) do if ch:IsA("TextButton") then ch:Destroy() end end
				for i, opt in ipairs(options) do
					local ob = new("TextButton", {
						Text = tostring(opt), Font = FONT, TextSize = TSIZE, AutoButtonColor = false,
						TextColor3 = (opt == selected) and C.accent or C.text, TextXAlignment = LEFT,
						BackgroundColor3 = C.hover, BackgroundTransparency = 1, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 18), LayoutOrder = i,
					}, list)
					new("UIPadding", { PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6) }, ob)
					ob.MouseEnter:Connect(function() ob.BackgroundTransparency = 0 end)
					ob.MouseLeave:Connect(function() ob.BackgroundTransparency = 1 end)
					ob.MouseButton1Click:Connect(function()
						selected = opt
						shown.Text = tostring(opt)
						list.Visible = false
						arrow.Text = "v"
						fire(o.Callback, opt)
					end)
				end
				list.Size = UDim2.new(1, 0, 0, math.max(1, math.min(#options, 6)) * 18)
			end
			rebuild()

			head.MouseButton1Click:Connect(function()
				if type(o.DynamicOptions) == "function" then options = o.DynamicOptions() end
				if #options == 0 then return end
				list.Visible = not list.Visible
				arrow.Text = list.Visible and "^" or "v"
				if list.Visible then rebuild() end
			end)
		end

		return tab
	end
	return Window
end

local function getHum()
	local c = player.Character
	return c and c:FindFirstChildOfClass("Humanoid")
end

local walkSpeed, jumpPower, flySpeed = 16, 50, 50
local noclip, infJump, godMode = false, false, false

local function applyStats()
	local hum = getHum()
	if hum then
		hum.UseJumpPower = true
		hum.WalkSpeed = walkSpeed
		hum.JumpPower = jumpPower
	end
end
player.CharacterAdded:Connect(function() task.wait(0.5) applyStats() end)

local flyConn, bv, bg
local function stopFly()
	if flyConn then flyConn:Disconnect() flyConn = nil end
	if bv then bv:Destroy() bv = nil end
	if bg then bg:Destroy() bg = nil end
	local hum = getHum()
	if hum then hum.PlatformStand = false end
end

local function startFly()
	local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	local hum = getHum()
	if not root or not hum then return end
	hum.PlatformStand = true
	bv = Instance.new("BodyVelocity", root)
	bv.MaxForce = Vector3.one * 1e9
	bv.Velocity = Vector3.zero
	bg = Instance.new("BodyGyro", root)
	bg.MaxTorque = Vector3.one * 1e9
	bg.P = 9e4
	bg.CFrame = root.CFrame
	
	flyConn = RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		local dir = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.yAxis end
		if dir.Magnitude == 0 and hum.MoveDirection.Magnitude > 0 then dir = hum.MoveDirection end
		bv.Velocity = dir.Magnitude > 0 and dir.Unit * flySpeed or Vector3.zero
		bg.CFrame = cam.CFrame
	end)
end

RunService.Stepped:Connect(function()
	if noclip and player.Character then
		for _, p in ipairs(player.Character:GetDescendants()) do
			if p:IsA("BasePart") then p.CanCollide = false end
		end
	end
end)

UserInputService.JumpRequest:Connect(function()
	if infJump then
		local hum = getHum()
		if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
	end
end)

	if fullbrightOn then
		Lighting.Ambient = Color3.new(1, 1, 1)
		Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
		Lighting.Brightness = 2
		Lighting.ClockTime = 14
		Lighting.GlobalShadows = false
	end
end)

-- Asset Retrieval Helpers
local function getFolderItems(pathTable, filterFunc)
	local items = {}
	local cur = ReplicatedStorage
	for _, p in ipairs(pathTable) do cur = cur and cur:FindFirstChild(p) end
	if cur then
		for _, ch in ipairs(cur:GetChildren()) do
			if not filterFunc or filterFunc(ch.Name) then table.insert(items, ch.Name) end
		end
	end
	return items
end

local function findItemDeep(root, name)
	if not root then return nil end
	if root:FindFirstChild(name) then return root:FindFirstChild(name) end
	for _, d in ipairs(root:GetDescendants()) do
		if d.Name:lower() == name:lower() then return d end
	end
	return nil
end

local function applyTrail(name)
	if not name then return notify("ToH", "Select a trail!") end

local function applyBillboard(name)

	local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")

		local template = nil

		if not template then
			for _, v in ipairs(ReplicatedStorage:GetDescendants()) do if v:IsA("BillboardGui") and v.Name == name then template = v break end end
		end

			bb.Size = UDim2.fromOffset(120, 40)
			bb.AlwaysOnTop = true

		bb.Adornee = hrp
		bb.StudsOffset = Vector3.new(0, 3, 0)
		bb.Parent = hrp

-- TTOH Automation
local ttohlRunning, cancelTTOH = false, false
local function runTTOH()
	if ttohlRunning then return end
	ttohlRunning = true
	cancelTTOH = false
	local stops, starts = {}, {}
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") then
			if d.Name:lower() == "stop" then table.insert(stops, d)
			elseif d.Name:lower() == "start" then table.insert(starts, d) end
		end
	end
	local count = 0
	for i = 1, math.max(#stops, #starts) do
		for _, p in ipairs({ stops[i], starts[i] }) do
			if cancelTTOH then break end
			local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
			if p and hrp then hrp.CFrame = p.CFrame + Vector3.new(0, 4, 0) count += 1 task.wait(0.25) end
		end
		if cancelTTOH then break end
	end
	ttohlRunning = false
	notify("TTOH", cancelTTOH and "Cancelled TTOH!" or ("Teleported to " .. count .. " parts."))
end
